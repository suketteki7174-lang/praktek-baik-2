-- Praktik Baik Dana Desa: akses tanpa login (role anon) -- Catatan: siapa pun yang mengetahui URL aplikasi dapat membaca/menambah/mengubah data dan mengunggah media. -- Penghapusan melalui API tidak diizinkan. Jangan simpan data pribadi/sensitif. -- Pastikan kolom yang dipakai aplikasi tersedia. ALTER TABLE public.kegiatan ADD COLUMN IF NOT EXISTS kode_desa text, ADD COLUMN IF NOT EXISTS desa text, ADD COLUMN IF NOT EXISTS titik integer, ADD COLUMN IF NOT EXISTS data jsonb, ADD COLUMN IF NOT EXISTS media jsonb NOT NULL DEFAULT '[]'::jsonb, ADD COLUMN IF NOT EXISTS updated_at timestamptz DEFAULT now(); CREATE UNIQUE INDEX IF NOT EXISTS kegiatan_kode_desa_titik_uidx ON public.kegiatan (kode_desa, titik); -- Hapus kebijakan ALL lama yang juga mengizinkan DELETE. DROP POLICY IF EXISTS "akses_kegiatan_aplikasi" ON public.kegiatan; DROP POLICY IF EXISTS "PB kegiatan public select" ON public.kegiatan; DROP POLICY IF EXISTS "PB kegiatan public insert" ON public.kegiatan; DROP POLICY IF EXISTS "PB kegiatan public update" ON public.kegiatan; CREATE POLICY "PB kegiatan public select" ON public.kegiatan FOR SELECT TO anon, authenticated USING (true); CREATE POLICY "PB kegiatan public insert" ON public.kegiatan FOR INSERT TO anon, authenticated WITH CHECK (true); CREATE POLICY "PB kegiatan public update" ON public.kegiatan FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true); -- Bucket publik agar URL foto/video dapat ditampilkan di aplikasi. INSERT INTO storage.buckets (id, name, public) VALUES ('dokumentasi', 'dokumentasi', true) ON CONFLICT (id) DO UPDATE SET public = true; DROP POLICY IF EXISTS "PB dokumentasi public read" ON storage.objects; DROP POLICY IF EXISTS "PB dokumentasi public upload" ON storage.objects; DROP POLICY IF EXISTS "PB dokumentasi public update" ON storage.objects; DROP POLICY IF EXISTS "PB dokumentasi public delete" ON storage.objects; CREATE POLICY "PB dokumentasi public read" ON storage.objects FOR SELECT TO anon, authenticated USING (bucket_id = 'dokumentasi'); CREATE POLICY "PB dokumentasi public upload" ON storage.objects FOR INSERT TO anon, authenticated WITH CHECK (bucket_id = 'dokumentasi'); -- Diperlukan untuk upload ulang dengan upsert:true pada nama file yang sama. CREATE POLICY "PB dokumentasi public update" ON storage.objects FOR UPDATE TO anon, authenticated USING (bucket_id = 'dokumentasi') WITH CHECK (bucket_id = 'dokumentasi'); -- Sengaja TIDAK membuat policy DELETE untuk tabel kegiatan maupun storage.objects.
